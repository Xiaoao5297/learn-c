# 6.4 链表进阶

6.3 的链表能建、能插、能遍历、能释放。这一节补上最后几块拼图——查找和删除，再把所有操作组装成一个完整的成绩单程序。思路很接近日常的成绩公示：名单插进去时按分数找好位置，打印就是从头到尾念一遍，查找、删除、释放也各有对应的生活动作。

## 6.4.1 查找

链表没有下标，查找只能从头顺着 `next` 逐个走——寻宝游戏里想确认某条线索存不存在，只能从第一张纸条开始一张张翻。找到第一个匹配的节点就返回它的地址，找不到返回 `NULL`：

```c
struct node *find_node(struct node *head, int target)
{
	for (struct node *cur = head; cur != NULL; cur = cur->next) {
		if (cur->data == target) {
			return cur;
		}
	}
	return NULL;
}
```

```c
// 6.4.1 list_find.c
#include <stdio.h>
#include <stdlib.h>

struct node {
	int data;
	struct node *next;
};

struct node *create_node(int value)
{
	struct node *n = malloc(sizeof(struct node));
	if (n == NULL) {
		return NULL;
	}
	n->data = value;
	n->next = NULL;
	return n;
}

void push_back(struct node **head, int value)
{
	struct node *n = create_node(value);
	if (n == NULL) {
		return;
	}

	if (*head == NULL) {
		*head = n;
		return;
	}

	struct node *tail = *head;
	while (tail->next != NULL) {
		tail = tail->next;
	}
	tail->next = n;
}

struct node *find_node(struct node *head, int target)
{
	for (struct node *cur = head; cur != NULL; cur = cur->next) {
		if (cur->data == target) {
			return cur;
		}
	}
	return NULL;
}

void free_list(struct node *head)
{
	struct node *cur = head;
	while (cur != NULL) {
		struct node *next = cur->next;
		free(cur);
		cur = next;
	}
}

int main(void)
{
	struct node *head = NULL;

	int values[] = {10, 20, 30};
	for (int i = 0; i < 3; i++) {
		push_back(&head, values[i]);
	}

	struct node *found = find_node(head, 20);
	if (found != NULL) {
		printf("找到 20\n");
	} else {
		printf("没找到 20\n");
	}

	found = find_node(head, 99);
	if (found == NULL) {
		printf("没找到 99\n");
	}

	free_list(head);
	head = NULL;

	return 0;
}
```

<!-- output: 6.4.1 -->
```text
找到 20
没找到 99
```

节点地址每次运行都不一样，所以示例不打印地址，只报告找没找到。空链表时 `find_node` 的循环一次都不进，直接返回 `NULL`——空链表安全。

查找的代价：最坏要走完全部 n 个节点；而数组 `arr[i]` 一步到位。经常"按位置取第几个"的场景，数组更合适。

## 6.4.2 删除节点

删除链表里第一个值为 target 的节点，比插入多一步想清楚：删掉一个节点后，断口要接回去。寻宝游戏里抽走一张纸条，前后两张的"位置线索"就断了，必须把前一张背面的地址改成指向后一张，游戏才能继续玩下去。

删除分三种情况：

```text
情况 1：空链表                无事可做
情况 2：删的是第一个节点      head 要改指第二个
情况 3：删的是中间或末尾节点  让前一个节点跳过目标
```

```c
int delete_node(struct node **head, int target)
{
	if (*head == NULL) {            /* 情况 1：空链表 */
		return 0;
	}

	if ((*head)->data == target) {  /* 情况 2：目标是头节点 */
		struct node *to_delete = *head;
		*head = (*head)->next;      /* head 指向第二个 */
		free(to_delete);
		return 1;
	}

	/* 情况 3：从头找"下一个是目标"的节点 prev */
	struct node *prev = *head;
	while (prev->next != NULL && prev->next->data != target) {
		prev = prev->next;
	}

	if (prev->next == NULL) {       /* 走到底也没找到 */
		return 0;
	}

	struct node *to_delete = prev->next;
	prev->next = to_delete->next;   /* prev 跳过目标，接上后面 */
	free(to_delete);
	return 1;
}
```

图解删除中间的 20（`10 -> 20 -> 30`）：

```text
删除前   head ──► 10 ──► 20 ──► 30 ──► NULL
                     prev   目标

prev->next = to_delete->next
删除后   head ──► 10 ──────────────► 30 ──► NULL
                          （20 已 free）
```

三个要点：

1. **删头节点会改 head**，所以参数还是 `struct node **`。删头的情况单独处理：先 `*head = (*head)->next` 接好，再 free 旧的第一个。
2. **先接指针，再 free**。顺序反了，`to_delete->next` 就读不到了——纸条撕掉之前没看清背面写的地址，撕完就再也接不回去了。
3. 情况 3 用"找前一个节点"的循环：条件 `prev->next != NULL && prev->next->data != target` 停在"下一个正好是目标"的位置，不用另设指针记前驱。

返回值用 `int`（删到了 1，没找到 0），让调用者知道删没删成，而不是闷声不响。

## 6.4.3 删除的完整演示

```c
// 6.4.3 list_delete.c
#include <stdio.h>
#include <stdlib.h>

struct node {
	int data;
	struct node *next;
};

struct node *create_node(int value)
{
	struct node *n = malloc(sizeof(struct node));
	if (n == NULL) {
		return NULL;
	}
	n->data = value;
	n->next = NULL;
	return n;
}

void push_back(struct node **head, int value)
{
	struct node *n = create_node(value);
	if (n == NULL) {
		return;
	}

	if (*head == NULL) {
		*head = n;
		return;
	}

	struct node *tail = *head;
	while (tail->next != NULL) {
		tail = tail->next;
	}
	tail->next = n;
}

void print_list(struct node *head)
{
	for (struct node *cur = head; cur != NULL; cur = cur->next) {
		printf("%d -> ", cur->data);
	}
	printf("NULL\n");
}

int delete_node(struct node **head, int target)
{
	if (*head == NULL) {
		return 0;
	}

	if ((*head)->data == target) {
		struct node *to_delete = *head;
		*head = (*head)->next;
		free(to_delete);
		return 1;
	}

	struct node *prev = *head;
	while (prev->next != NULL && prev->next->data != target) {
		prev = prev->next;
	}

	if (prev->next == NULL) {
		return 0;
	}

	struct node *to_delete = prev->next;
	prev->next = to_delete->next;
	free(to_delete);
	return 1;
}

void free_list(struct node *head)
{
	struct node *cur = head;
	while (cur != NULL) {
		struct node *next = cur->next;
		free(cur);
		cur = next;
	}
}

int main(void)
{
	struct node *head = NULL;

	int values[] = {10, 20, 30, 40};
	for (int i = 0; i < 4; i++) {
		push_back(&head, values[i]);
	}
	printf("初始链表：");
	print_list(head);

	delete_node(&head, 20);
	printf("删除 20 后：");
	print_list(head);

	delete_node(&head, 10);
	printf("删除 10（头节点）后：");
	print_list(head);

	if (delete_node(&head, 99) == 0) {
		printf("没找到 99，链表不变：");
		print_list(head);
	}

	free_list(head);
	head = NULL;

	return 0;
}
```

## 6.4.4 综合项目：成绩单链表

把本章所有操作组装成一个完整的程序：`struct student` 链表，支持按分数**有序插入**、打印、查找、删除、释放。

```c
// 6.4.4 score_list.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NAME_LEN 20

struct student {
	char name[NAME_LEN];
	double score;
	struct student *next;
};

struct student *create_student(const char *name, double score)
{
	struct student *s = malloc(sizeof(struct student));
	if (s == NULL) {
		return NULL;
	}

	snprintf(s->name, sizeof(s->name), "%s", name);
	s->score = score;
	s->next = NULL;
	return s;
}

/* 按分数从高到低插入；插入失败返回 0 */
int insert_sorted(struct student **head, const char *name, double score)
{
	struct student *s = create_student(name, score);
	if (s == NULL) {
		return 0;
	}

	/* 空链表，或新学生分数比第一名还高：插到最前面 */
	if (*head == NULL || (*head)->score < score) {
		s->next = *head;
		*head = s;
		return 1;
	}

	/* 找到"下一个分数更低"的位置，插进去 */
	struct student *cur = *head;
	while (cur->next != NULL && cur->next->score >= score) {
		cur = cur->next;
	}
	s->next = cur->next;
	cur->next = s;
	return 1;
}

void print_list(const struct student *head)
{
	int rank = 1;
	for (const struct student *cur = head; cur != NULL; cur = cur->next) {
		printf("第%d名：%s（%.1f 分）\n", rank, cur->name, cur->score);
		rank++;
	}
}

struct student *find_student(struct student *head, const char *name)
{
	for (struct student *cur = head; cur != NULL; cur = cur->next) {
		if (strcmp(cur->name, name) == 0) {
			return cur;
		}
	}
	return NULL;
}

int delete_student(struct student **head, const char *name)
{
	if (*head == NULL) {
		return 0;
	}

	if (strcmp((*head)->name, name) == 0) {
		struct student *to_delete = *head;
		*head = (*head)->next;
		free(to_delete);
		return 1;
	}

	struct student *prev = *head;
	while (prev->next != NULL && strcmp(prev->next->name, name) != 0) {
		prev = prev->next;
	}

	if (prev->next == NULL) {
		return 0;
	}

	struct student *to_delete = prev->next;
	prev->next = to_delete->next;
	free(to_delete);
	return 1;
}

void free_list(struct student *head)
{
	struct student *cur = head;
	while (cur != NULL) {
		struct student *next = cur->next;
		free(cur);
		cur = next;
	}
}

int main(void)
{
	struct student *head = NULL;

	/* 乱序插入，insert_sorted 自动按分数排好 */
	const char *names[] = {"xiaoming", "xiaohong", "xiaogang", "xiaoli", "xiaohua"};
	double scores[] = {85.5, 92.0, 76.5, 98.0, 60.0};
	for (int i = 0; i < 5; i++) {
		if (!insert_sorted(&head, names[i], scores[i])) {
			printf("创建节点失败\n");
			free_list(head);
			return 1;
		}
	}

	printf("=== 成绩单 ===\n");
	print_list(head);

	struct student *found = find_student(head, "xiaohong");
	if (found != NULL) {
		printf("查到 xiaohong：%.1f 分\n", found->score);
	}

	if (delete_student(&head, "xiaohong") == 1) {
		printf("删除 xiaohong 后：\n");
		print_list(head);
	}

	free_list(head);
	head = NULL;

	return 0;
}
```

```text
第1名：xiaoli（98.0 分）
第2名：xiaohong（92.0 分）
第3名：xiaoming（85.5 分）
第4名：xiaogang（76.5 分）
第5名：xiaohua（60.0 分）
查到 xiaohong：92.0 分
```

这个程序里藏着几章积累的成果：

1. **有序插入**就是"链表版插入排序"：每次找对位置、改两个指针，不像数组那样要挪动一串元素——新同学插队按成绩站位时，不用全班起立挪座位，只需前后两位改一下"我后面是谁"——这正是链表"增删快"的体现。
2. **名字拷贝用 `snprintf`**：它最多写 `sizeof(s->name)` 字节并保证 `'\0'` 结尾，超长名字自动截断。第 3 章讲过的"不用 `strcpy`"在这里直接用上。
3. **每个函数都空链表安全**：`insert_sorted` 特判 `*head == NULL`，`print_list`、`find_student`、`delete_student` 的循环遇到 `NULL` 自然结束。
4. **创建失败也处理**：`insert_sorted` 把 `create_student` 的 `NULL` 如实上报，`main` 收到后释放已建好的链再退出——失败路径同样不忘 `free`。

## 6.4.5 链表能存进文件吗

第 5 章学过把结构体数组整块 `fwrite` 进文件。链表能不能照搬？**不能直接搬**。节点的 `next` 存的是内存地址，这次运行和下次运行的地址毫无关系，把地址写进文件再读回来全是废数据——寻宝游戏的纸条背面写的是"我家的门牌号"，游戏换个场地重开，门牌号就全作废了，能带走的只有纸条正面写的宝贝（数据）。

正确做法是**只存数据，读回来重新搭链**：

```c
/* 存盘：只写 name 和 score，绝不写 next */
int save_students(const struct student *head, const char *filename)
{
	FILE *fp = fopen(filename, "w");
	if (fp == NULL) {
		return -1;
	}

	for (const struct student *cur = head; cur != NULL; cur = cur->next) {
		fprintf(fp, "%s %.1f\n", cur->name, cur->score);
	}

	fclose(fp);
	return 0;
}

/* 读盘：每读一行就重新建一个节点，重新搭链 */
int load_students(struct student **head, const char *filename)
{
	FILE *fp = fopen(filename, "r");
	if (fp == NULL) {
		return -1;
	}

	char name[NAME_LEN];
	double score;
	while (fscanf(fp, "%19s %lf", name, &score) == 2) {
		insert_sorted(head, name, score);
	}

	fclose(fp);
	return 0;
}
```

一句话：**结构体可以整块存取，链表不行——文件里只放数据，指针是运行时临时搭的。** 换句话说，带走的只有宝贝清单，纸条间的线索要等游戏重新开始时现场补写。读入用的 `%19s` 限宽、`fscanf` 返回值检查，都是第 5 章的老规矩。

## 6.4.6 常见错误

| 错误写法 | 问题 | 正确写法 |
|----------|------|----------|
| `delete_node` 只收 `struct node *` | 删头节点时改不动外面的 head | 收 `struct node **`，删头改 `*head` |
| 先 `free(to_delete)` 再读 `to_delete->next` | 悬垂指针 | `prev->next = to_delete->next;` 之后才 `free` |
| 删头节点后直接 `free(*head)` | head 变成悬垂指针 | 先 `*head = (*head)->next` 接好再 free |
| 文件里存整个节点（含 `next`） | 读回来地址全废 | 只存数据，读回来重建链 |
| `insert_sorted` 不特判空链表 | 对 `NULL` 解引用 | 先判断 `*head == NULL` 再比较分数 |
| 删除、查找失败不报告 | 调用者不知道结果 | 返回 `1`/`0` 或节点/`NULL` |

## 6.4.7 小结

- 查找从头逐个走，找到返回节点地址，找不到返回 `NULL`；代价是 O(n)。
- 删除分三种情况：空链表直接返回、删头节点要改 `*head`、删中间让前驱跳过目标。
- 顺序铁律：先接好指针，再 free。
- 综合程序 = 创建 + 有序插入 + 遍历 + 查找 + 删除 + 释放，每个函数都空链表安全，分配失败如实上报。
- 链表落盘只存数据不存指针，读回来重新搭链。
- 数组"查得快、增删慢"，链表"增删快、查得慢"——按场景选结构，是数据结构学习的起点。

## 练习

### 一、选择题

**1. 删除头节点和删除中间节点，处理上的关键区别是什么？**

A. 没有区别，处理方式完全一样
B. 删头节点要额外更新 `head`（通常通过`*head`），否则整个链表跟着消失；删中间节点只需让前驱的`next` 跳过目标
C. 删中间节点要更新 `head`，删头节点不用
D. 两个都不需要移动指针

**2. `delete_node` 里为什么必须先写`prev->next = to_delete->next;` 再`free(to_delete);`？**

A. 顺序无所谓，编译器会调整
B. `free` 之后再读`to_delete->next` 是访问已释放的内存（悬垂访问），结果不确定
C. `free` 会把内存清零，先释放就拿不到`next` 的值了
D. `free` 要求函数的第一条语句必须是它

**3. 为什么链表不能像结构体数组那样整块 `fwrite` 存盘？**

A. `fwrite` 不支持指针类型
B. 节点里的 `next` 存的是内存地址，只在本程序这一次运行中有意义，写进文件再读回来链是断的
C. 链表节点太大，`fwrite` 写不下
D. 可以存，只是速度慢

#### 一、选择题

1. B。头节点是链表的"入口"，删它必须把 `head` 挪到第二个节点；中间节点只要接好前后两截。
2. B。`to_delete->next` 是接上断口需要的信息，必须先抄走再释放；先释放再读就是悬垂访问。
3. B。地址只在这次运行中有意义，下次运行内存布局完全不同，存地址等于存了一串没用的数。

#### 二、填空题

4. `next`（后继）。
5. 内存地址。

#### 三、改错题

6. 三处错误：

- 第一处：参数是 `struct node *head`，删头节点时即使`free` 了，调用者的头指针也不动，链表照旧指向已释放的内存。参数应为`struct node **head`。
- 第二处：删头节点时直接 `free(head)` 就返回，没有把头指针接到第二个节点上，链表整个断掉。应先把`*head` 指向`head->next` 再释放。
- 第三处：返回值是 `void`，调用者无从知道删没删成。应返回`int`：删到返回 1，没找到返回 0（`main` 里传的`&a` 也要与`struct node **` 参数匹配）。

修正后的完整程序：

```c
#include <stdio.h>
#include <stdlib.h>

struct node {
	int data;
	struct node *next;
};

int delete_node(struct node **head, int target)
{
	if (*head == NULL) {
		return 0;
	}

	if ((*head)->data == target) {
		struct node *to_delete = *head;
		*head = to_delete->next;
		free(to_delete);
		return 1;
	}

	struct node *prev = *head;
	while (prev->next != NULL && prev->next->data != target) {
		prev = prev->next;
	}

	struct node *to_delete = prev->next;
	if (to_delete == NULL) {
		return 0;
	}
	prev->next = to_delete->next;
	free(to_delete);
	return 1;
}

struct node *create_node(int value)
{
	struct node *n = malloc(sizeof(struct node));
	if (n == NULL) {
		return NULL;
	}
	n->data = value;
	n->next = NULL;
	return n;
}

int main(void)
{
	struct node *head = NULL;
	struct node *tail = NULL;
	int values[] = {1, 2, 3};

	for (int i = 0; i < 3; i++) {
		struct node *n = create_node(values[i]);
		if (n == NULL) {
			return 1;
		}
		if (head == NULL) {
			head = n;
		} else {
			tail->next = n;
		}
		tail = n;
	}

	delete_node(&head, 2);

	printf("%d -> %d\n", head->data, head->next->data);

	return 0;
}
```

输出 `1 -> 3`。注意节点要用`malloc` 申请：`free` 只能释放动态内存，对栈上的变量调用`free` 是未定义行为。

#### 四、编程题

7. 参考代码：

```c
#include <stdio.h>

struct node {
	int data;
	struct node *next;
};

int find_node(struct node *head, int target)
{
	for (struct node *cur = head; cur != NULL; cur = cur->next) {
		if (cur->data == target) {
			return 1;
		}
	}
	return 0;
}

int main(void)
{
	struct node c = {30, NULL};
	struct node b = {20, &c};
	struct node a = {10, &b};

	if (find_node(&a, 20)) {
		printf("查找 20：找到了\n");
	} else {
		printf("查找 20：没找到\n");
	}

	if (find_node(&a, 99)) {
		printf("查找 99：找到了\n");
	} else {
		printf("查找 99：没找到\n");
	}

	return 0;
}
```

输出：

```text
查找 99：没找到
```

8. 参考代码：

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NAME_LEN 20

struct student {
	char name[NAME_LEN];
	double score;
	struct student *next;
};

struct student *create_student(const char *name, double score)
{
	struct student *s = malloc(sizeof(struct student));
	if (s == NULL) {
		return NULL;
	}
	strcpy(s->name, name);
	s->score = score;
	s->next = NULL;
	return s;
}

void push_back(struct student **head, const char *name, double score)
{
	struct student *s = create_student(name, score);
	if (s == NULL) {
		return;
	}

	if (*head == NULL) {
		*head = s;
		return;
	}

	struct student *tail = *head;
	while (tail->next != NULL) {
		tail = tail->next;
	}
	tail->next = s;
}

int delete_student(struct student **head, const char *name)
{
	if (*head == NULL) {
		return 0;
	}

	if (strcmp((*head)->name, name) == 0) {
		struct student *to_delete = *head;
		*head = to_delete->next;
		free(to_delete);
		return 1;
	}

	struct student *prev = *head;
	while (prev->next != NULL && strcmp(prev->next->name, name) != 0) {
		prev = prev->next;
	}

	if (prev->next == NULL) {
		return 0;
	}

	struct student *to_delete = prev->next;
	prev->next = to_delete->next;
	free(to_delete);
	return 1;
}

int list_length(struct student *head)
{
	int count = 0;
	for (struct student *cur = head; cur != NULL; cur = cur->next) {
		count++;
	}
	return count;
}

void free_list(struct student *head)
{
	struct student *cur = head;
	while (cur != NULL) {
		struct student *next = cur->next;
		free(cur);
		cur = next;
	}
}

int main(void)
{
	struct student *head = NULL;

	push_back(&head, "xiaoming", 90.0);
	push_back(&head, "xiaohong", 85.0);
	push_back(&head, "xiaogang", 78.0);
	push_back(&head, "xiaoli", 92.0);
	push_back(&head, "xiaozhang", 66.0);

	printf("删除前人数：%d\n", list_length(head));

	delete_student(&head, "xiaohong");

	printf("删除后人数：%d\n", list_length(head));

	free_list(head);
	head = NULL;

	return 0;
}
```

输出：

```text
删除前人数：5
删除后人数：4
```

#### 五、思考题

9. 开放性引导，参考方向：链表天然是"递归形状"的——一个节点加一条更短的链表；函数必须有基线条件（这里是空链表 `NULL`）才能停下；链表长度完全可以写成"1 + 剩下链表的长度"。下一章就从这样的公式开始，正式认识递归。
