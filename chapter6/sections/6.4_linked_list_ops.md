## 6.4.1 查找：find / contains

数组用下标找很快，链表只能**从头挨个找**。

### 📝 查找某个值，返回第一个匹配的节点

```c
// 在链表里找第一个 data == target 的节点；找不到返回 NULL
Node *findNode(Node *head, int target) {
    for (Node *cur = head; cur != NULL; cur = cur->next) {
        if (cur->data == target) {
            return cur;          // 找到了，返回节点地址
        }
    }
    return NULL;                 // 没找到
}
```

### 📝 用法

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node *createNode(int value) { /* 同上一节 */ }
void push_back(Node **head, int value) { /* 同上一节 */ }
void printList(Node *head) { /* 同上一节 */ }
void freeList(Node *head) { /* 同上一节 */ }

// 查找
Node *findNode(Node *head, int target) {
    for (Node *cur = head; cur != NULL; cur = cur->next) {
        if (cur->data == target) {
            return cur;
        }
    }
    return NULL;
}

int main() {
    Node *head = NULL;
    push_back(&head, 10);
    push_back(&head, 20);
    push_back(&head, 30);
    printList(head);

    Node *found = findNode(head, 20);
    if (found != NULL) {
        printf("找到了 20！\n");
    } else {
        printf("没找到 20。\n");
    }

    found = findNode(head, 99);
    if (found != NULL) {
        printf("找到了 99！\n");
    } else {
        printf("没找到 99。\n");
    }

    freeList(head);
    return 0;
}
```

输出：
```
10 -> 20 -> 30 -> NULL
找到了 20！
没找到 99。
```

> 📌 **查找的代价：** 链表的查找必须从头走到尾，最坏要走 n 步（O(n)）。数组 `arr[i]` 一步到位（O(1)）。所以"经常按位置随机访问"的场景，数组更合适。

## 6.4.2 删除：delete_by_value

删除链表里第一个值等于 target 的节点。

### 🎯 先看两种情况的区别

```
情况A（删的是第一个节点）：
head → [20] → 30 → NULL
        删除后：head 要指向 30！

情况B（删的是中间/末尾节点）：
head → 10 → [20] → 30 → NULL
              删除后：让 10 的 next 跳过 20 指向 30
```

**删除的关键：找到"目标节点的前一个节点"，让它跳过目标。**

### 📝 实现

```c
// 删除第一个 data == target 的节点；删到了返回1，没删到返回0
int deleteNode(Node **head, int target) {
    // 情况1：链表空
    if (*head == NULL) {
        return 0;
    }
    
    // 情况2：目标正好是头节点 → 要改 *head
    if ((*head)->data == target) {
        Node *toDelete = *head;
        *head = (*head)->next;     // head 指向第二个
        free(toDelete);            // 释放被删节点
        return 1;
    }
    
    // 情况3：目标在中间或末尾 → 找到它的前一个节点 prev
    Node *prev = *head;
    while (prev->next != NULL && prev->next->data != target) {
        prev = prev->next;         // 往前走，直到 prev->next 是目标
    }
    
    if (prev->next == NULL) {      // 走到底都没找到
        return 0;
    }
    
    Node *toDelete = prev->next;   // 目标节点
    prev->next = toDelete->next;   // 让 prev 跳过目标，连到下一个
    free(toDelete);                // 释放
    return 1;
}
```

**图解"删除中间的 20"：**

```
删除前：head → 10 → [20] → 30 → NULL
                   prev   目标
prev->next = 目标的下一个
      ↓
删除后：head → 10 ──────────→ 30 → NULL
                  （20 已被 free）
```

> 🚨 **两个容易错的地方：**
> 1. **删头节点要改 `*head`**，所以参数是 `Node **`（老规矩：要改外面变量就传地址）
> 2. **先接好指针，再 free**，顺序不能反

## 6.4.3 求长度：list_length

数一数链表有多少个节点：

```c
int listLength(Node *head) {
    int count = 0;
    for (Node *cur = head; cur != NULL; cur = cur->next) {
        count++;
    }
    return count;
}
```

> 📌 链表没有"长度属性"，想知道的实时长度就得**走一遍数一遍**。这也是为什么用链表要尽量少"按位置取第几个"。

## 6.4.4 综合实战：学生成绩链表

把 6.2 的"动态数组学生名单"换成**链表版**——支持插入、打印、查找、删除：

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Student {
    char name[20];
    double score;
    struct Student *next;
} Student;

// 造一个学生节点
Student *createStudent(const char *name, double score) {
    Student *s = (Student *)malloc(sizeof(Student));
    if (s == NULL) return NULL;
    // 简单方式：先把名字清空再逐个拷贝（或直接用 sprintf 也行）
    int i = 0;
    while (name[i] != '\0' && i < 19) {
        s->name[i] = name[i];
        i++;
    }
    s->name[i] = '\0';
    s->score = score;
    s->next = NULL;
    return s;
}

// 按成绩从高到低插入（简化版：直接插到合适位置保持降序）
void insertSorted(Student **head, const char *name, double score) {
    Student *s = createStudent(name, score);
    if (s == NULL) return;
    
    // 空链表，或新成绩比当前第一名还高 → 插到最前面
    if (*head == NULL || (*head)->score < score) {
        s->next = *head;
        *head = s;
        return;
    }
    
    // 否则找到"下一个比我低"的位置插进去
    Student *cur = *head;
    while (cur->next != NULL && cur->next->score >= score) {
        cur = cur->next;
    }
    s->next = cur->next;
    cur->next = s;
}

void printList(Student *head) {
    if (head == NULL) {
        printf("（空名单）\n");
        return;
    }
    int rank = 1;
    for (Student *cur = head; cur != NULL; cur = cur->next) {
        printf("第%d名：%s（%.1f分）\n", rank, cur->name, cur->score);
        rank++;
    }
}

void freeList(Student *head) {
    Student *cur = head;
    while (cur != NULL) {
        Student *next = cur->next;
        free(cur);
        cur = next;
    }
}

int main() {
    Student *head = NULL;
    
    // 乱序插入，靠 insertSorted 自动按分数排序
    insertSorted(&head, "小明", 85.5);
    insertSorted(&head, "小红", 92.0);
    insertSorted(&head, "小刚", 76.5);
    insertSorted(&head, "小丽", 98.0);
    insertSorted(&head, "小华", 60.0);
    
    printf("=== 按成绩降序的名单 ===\n");
    printList(head);
    
    freeList(head);
    return 0;
}
```

输出：
```
=== 按成绩降序的名单 ===
第1名：小丽（98.0分）
第2名：小红（92.0分）
第3名：小明（85.5分）
第4名：小刚（76.5分）
第5名：小华（60.0分）
```

> 🧠 **看，链表版的插入排序就藏在这里！** 每次插入时"找对位置、改两个指针"，不用像数组那样挪动一堆元素——这正是链表的威力。

## 6.4.5 一个提醒：链表和文件能合体吗？

> 🤔 细心的你会发现：第5章我们学了"把结构体数组用 fwrite/fread 存进文件"。链表能存文件吗？
>
> **可以，但要小心！** 链表节点之间靠的是**内存地址**连接；存盘时地址毫无意义（下次运行地址全变了）。正确做法是：
> 1. 存盘：逐个节点把 `data` 写进文件（**别写 next 指针！**）
> 2. 读盘：边读边 `createStudent`/`push_back`，**重新搭链**
>
> 换句话说：**文件里只存"数据"，指针是运行时临时搭的。**

```c
// 存盘示例：只写数据
void saveToFile(Student *head, const char *filename) {
    FILE *fp = fopen(filename, "w");
    if (fp == NULL) { perror("保存失败"); return; }
    for (Student *cur = head; cur != NULL; cur = cur->next) {
        fprintf(fp, "%s %.1f\n", cur->name, cur->score);   // 只写数据
    }
    fclose(fp);
}

// 读盘示例：重建链表
void loadFromFile(Student **head, const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (fp == NULL) return;                 // 没有文件就当空名单
    char name[20];
    double score;
    while (fscanf(fp, "%19s %lf", name, &score) == 2) {
        insertSorted(head, name, score);    // 边读边重建
    }
    fclose(fp);
}
```

> 💡 **一句话：结构体可以直接 fwrite/fread；链表不行（有指针）。链表落盘要"拆开存数据、读回来重拼"。**

## 6.4.6 本节小结

这一节我们学了：

✅ **查找：** 从头顺着 next 找，找不到返回 NULL（O(n)）

✅ **删除：** 关键在找"前一个节点"，让它跳过目标；删头节点要改 `*head`

✅ **顺序别反：** 先接好指针，再 free

✅ **求长度：** 走一遍数一遍

✅ **综合应用：** 插入排序式的"按成绩排序名单"

✅ **链表与文件：** 只存数据不存指针，读回来重建

> 💡 **记住：链表操作 = 画箭头 + 改箭头 + 别忘了释放。数组靠下标，链表靠指针！**

### 三、改错题

**6. 以下程序想删除值为 20 的节点，有4个错误，请找出并改正：**

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

void push_back(Node **head, int value) { /* 假设正确实现 */ }
void freeList(Node *head) { /* 假设正确实现 */ }
void printList(Node *head) { /* 假设正确实现 */ }

void deleteNode(Node *head, int target) {   // 错误区①：想改 head 却只传了 Node*
    // 删头节点
    if (head->data == target) {             // 错误区②：应该用 (*head) 判断
        head = head->next;                  // 错误区③：改的是形参副本
        return;
    }
    // 删中间
    Node *prev = head;
    while (prev->next != NULL && prev->next->data != target) {
        prev = prev->next;
    }
    if (prev->next != NULL) {
        Node *toDelete = prev->next;
        prev->next = toDelete->next;
        // 错误区④：忘了 free(toDelete)
    }
}

int main() {
    Node *head = NULL;
    push_back(&head, 10);
    push_back(&head, 20);
    push_back(&head, 30);
    deleteNode(head, 20);       // 调用也应该传 &head
    printList(head);
    freeList(head);
    return 0;
}
```
