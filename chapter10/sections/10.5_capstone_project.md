## 10.5.2 设计数据结构

一个人需要"姓名 + 电话"，用结构体打包（第4章）：

```c
typedef struct {
    char name[32];     // 姓名
    char phone[20];    // 电话
} Contact;
```

再开一个数组装"很多人"，并用一个 `count` 记录当前有多少人（第3章思路）：

```c
#define MAX_CONTACTS 100
Contact book[MAX_CONTACTS];   // 通讯录本体
int count = 0;                // 当前人数
```

> 📌 `#define MAX_CONTACTS` 和 `#define FILE_NAME`（文件名）都是 10.1 的**宏**——想改容量/文件名，只改一行！

## 10.5.3 分模块实现（先每个都写通）

### 模块 A：添加 & 显示

```c
void addContact(void) {
    if (count >= MAX_CONTACTS) {        // 边界检查：满了就拒收
        printf("通讯录满了！\n");
        return;
    }
    printf("姓名：");
    scanf("%s", book[count].name);
    printf("电话：");
    scanf("%s", book[count].phone);
    printf("已添加：%s（%s）\n", book[count].name, book[count].phone);
    count++;                            // 别忘 +1
}

void listAll(void) {
    if (count == 0) {
        printf("通讯录是空的\n");
        return;
    }
    printf("序号  姓名        电话\n");
    printf("----------------------\n");
    for (int i = 0; i < count; i++) {
        printf("%-4d %-10s %s\n", i + 1, book[i].name, book[i].phone);
    }
}
```

> 💡 `%-4d`、`%-10s` 是"左对齐 + 占宽"的打印，让列表整整齐齐。

### 模块 B：查找 & 排序

```c
void searchByName(void) {
    char key[32];
    printf("要找谁：");
    scanf("%s", key);
    int found = 0;
    for (int i = 0; i < count; i++) {
        if (strcmp(book[i].name, key) == 0) {   // 字符串相等判断用 strcmp！
            printf("找到：%s（%s）\n", book[i].name, book[i].phone);
            found = 1;
        }
    }
    if (!found) {
        printf("没找到 %s\n", key);
    }
}

// qsort 的比较器：比姓名（10.4 现学现用）
int cmpByName(const void *a, const void *b) {
    const Contact *ca = (const Contact *)a;
    const Contact *cb = (const Contact *)b;
    return strcmp(ca->name, cb->name);
}

void sortByName(void) {
    qsort(book, count, sizeof(Contact), cmpByName);  // 一行排序！
    printf("已按姓名排序：\n");
    listAll();
}
```

> 🎉 想想第8章我们手写的冒泡排序有多长——这里 `qsort` 一行搞定，还不用自己写交换。**这就是函数指针 + 标准库的威力。**

### 模块 C：存盘 & 读盘

文件格式很简单：第一行人数，之后每行 `姓名 电话`：

```c
void saveFile(void) {
    FILE *fp = fopen(FILE_NAME, "w");
    if (fp == NULL) {                       // 铁律：打开文件必须检查！
        printf("保存失败！\n");
        return;
    }
    fprintf(fp, "%d\n", count);             // 先写人数
    for (int i = 0; i < count; i++) {
        fprintf(fp, "%s %s\n", book[i].name, book[i].phone);
    }
    fclose(fp);                             // 拿完要还（第5章）
    printf("已保存 %d 条到 %s\n", count, FILE_NAME);
}

void loadFile(void) {
    FILE *fp = fopen(FILE_NAME, "r");
    if (fp == NULL) {
        printf("没有可读取的文件\n");
        return;
    }
    fscanf(fp, "%d", &count);               // 读人数
    for (int i = 0; i < count; i++) {
        fscanf(fp, "%s %s", book[i].name, book[i].phone);
    }
    fclose(fp);
    printf("已从 %s 读入 %d 条\n", FILE_NAME, count);
}
```

## 10.5.4 完整源代码

把上面所有模块 + 菜单 `main` 拼在一起，就是完整项目：

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_CONTACTS 100
#define FILE_NAME "contacts.txt"

typedef struct {
    char name[32];
    char phone[20];
} Contact;

Contact book[MAX_CONTACTS];
int count = 0;

void addContact(void) {
    if (count >= MAX_CONTACTS) {
        printf("通讯录满了！\n");
        return;
    }
    printf("姓名：");
    scanf("%s", book[count].name);
    printf("电话：");
    scanf("%s", book[count].phone);
    printf("已添加：%s（%s）\n", book[count].name, book[count].phone);
    count++;
}

void listAll(void) {
    if (count == 0) {
        printf("通讯录是空的\n");
        return;
    }
    printf("序号  姓名        电话\n");
    printf("----------------------\n");
    for (int i = 0; i < count; i++) {
        printf("%-4d %-10s %s\n", i + 1, book[i].name, book[i].phone);
    }
}

void searchByName(void) {
    char key[32];
    printf("要找谁：");
    scanf("%s", key);
    int found = 0;
    for (int i = 0; i < count; i++) {
        if (strcmp(book[i].name, key) == 0) {
            printf("找到：%s（%s）\n", book[i].name, book[i].phone);
            found = 1;
        }
    }
    if (!found) {
        printf("没找到 %s\n", key);
    }
}

int cmpByName(const void *a, const void *b) {
    const Contact *ca = (const Contact *)a;
    const Contact *cb = (const Contact *)b;
    return strcmp(ca->name, cb->name);
}

void sortByName(void) {
    qsort(book, count, sizeof(Contact), cmpByName);
    printf("已按姓名排序：\n");
    listAll();
}

void saveFile(void) {
    FILE *fp = fopen(FILE_NAME, "w");
    if (fp == NULL) {
        printf("保存失败！\n");
        return;
    }
    fprintf(fp, "%d\n", count);
    for (int i = 0; i < count; i++) {
        fprintf(fp, "%s %s\n", book[i].name, book[i].phone);
    }
    fclose(fp);
    printf("已保存 %d 条到 %s\n", count, FILE_NAME);
}

void loadFile(void) {
    FILE *fp = fopen(FILE_NAME, "r");
    if (fp == NULL) {
        printf("没有可读取的文件\n");
        return;
    }
    fscanf(fp, "%d", &count);
    for (int i = 0; i < count; i++) {
        fscanf(fp, "%s %s", book[i].name, book[i].phone);
    }
    fclose(fp);
    printf("已从 %s 读入 %d 条\n", FILE_NAME, count);
}

int main(void) {
    int choice;
    while (1) {
        printf("\n====== 我的通讯录 ======\n");
        printf("1. 添加联系人\n");
        printf("2. 显示全部\n");
        printf("3. 按姓名查找\n");
        printf("4. 按姓名排序\n");
        printf("5. 保存到文件\n");
        printf("6. 从文件读取\n");
        printf("0. 退出\n");
        printf("请选择：");
        scanf("%d", &choice);

        switch (choice) {
            case 1:  addContact();  break;
            case 2:  listAll();     break;
            case 3:  searchByName(); break;
            case 4:  sortByName();  break;
            case 5:  saveFile();    break;
            case 6:  loadFile();    break;
            case 0:
                printf("再见！数据已保存在内存中，记得先选 5 存盘哦。\n");
                return 0;
            default:
                printf("没有这个选项，请重试\n");
        }
    }
    return 0;
}
```

> 📌 **`main` 就是一个"总指挥"：** `while(1)` 死循环保持菜单不退出；`scanf("%d", &choice)` 读选项；`switch` 分发任务；选 `0` 时 `return 0` 退出。你看，全是第2章的老朋友！
