# 10.5 毕业项目：通讯录

从第 1 章的第一行代码到上一节的比较器，零件已经攒齐了。这一节把它们全部装进同一个程序——**通讯录**。

它能：添加联系人、显示全部、按姓名查找、按姓名排序、删除联系人、保存到文件、从文件读取，退出时自动存盘。功能清单里每个词背后都是学过的一章：

| 选项 | 功能 | 用到的知识 |
|------|------|-----------|
| 1 | 添加联系人 | 结构体、限宽 `scanf`、检查返回值（第 3 章） |
| 2 | 显示全部 | 数组遍历、格式化打印（第 3 章） |
| 3 | 按姓名查找 | `strcmp`（第 3 章）、线性查找（第 8 章） |
| 4 | 按姓名排序 | `qsort` + 比较器（10.4） |
| 5 | 删除联系人 | 数组删除、结构体赋值（第 3、4 章） |
| 6/7 | 存盘/读盘 | `fopen`/`fprintf`/`fscanf`、校验文件头（第 5 章） |
| 0 | 退出 | 菜单循环（第 2 章）、退出自动保存 |
| 菜单本身 | 函数指针数组分发 | 10.3 |

这个项目的骨架和真实的小型管理系统一模一样：**数据结构 + 增删查改 + 持久化 + 菜单交互**。做完它，这本教程就毕业了。

## 10.5.1 数据结构：一个人怎么装

一个人需要"姓名 + 电话"，用结构体打包（第 4 章的老朋友）：

```c
struct contact {
	char name[30];
	char phone[20];
};
```

按本书规范用 `struct contact`，不用 `typedef`（2.6.9）。容量和文件名用常量宏（10.1），想改容量只改一行：

```c
#define MAX_CONTACTS 100
#define NAME_LEN 30
#define PHONE_LEN 20
#define DATA_FILE "contacts_book.txt"
```

通讯录本体是一个结构体数组，配一个 `count` 记录当前人数：

```c
struct contact book[MAX_CONTACTS];
int count = 0;
```

数组和 `count` 用全局变量，是为了让示例代码聚焦在"功能怎么写"上；4.7 的多文件工程里，它们会被装进 `.c` 文件内部、用函数接口暴露。字符数组的尺寸按 3.3.6 的估算：姓名留 29 个字节，中文姓名 9 个字也放得下；电话留 19 个字节。

## 10.5.2 里程碑一：添加与显示

第一个里程碑只有两个功能：添加、显示。写完就跑，跑通了再加下一个。

```c
// 10.5.1 contact_milestone1.c
#include <stdio.h>
#include <string.h>

#define MAX_CONTACTS 5
#define NAME_LEN 30
#define PHONE_LEN 20

struct contact {
	char name[NAME_LEN];
	char phone[PHONE_LEN];
};

struct contact book[MAX_CONTACTS];
int count = 0;

int add_contact(void)
{
	if (count >= MAX_CONTACTS) {
		printf("通讯录已满\n");
		return -1;
	}

	printf("请输入姓名：");
	if (scanf("%29s", book[count].name) != 1) {
		printf("姓名读取失败\n");
		return -1;
	}

	printf("请输入电话：");
	if (scanf("%19s", book[count].phone) != 1) {
		printf("电话读取失败\n");
		return -1;
	}

	printf("已添加：%s（%s）\n", book[count].name, book[count].phone);
	count++;
	return 0;
}

void list_contacts(void)
{
	if (count == 0) {
		printf("通讯录是空的\n");
		return;
	}

	printf("共 %d 人：\n", count);
	for (int i = 0; i < count; i++) {
		printf("%d. %s %s\n", i + 1, book[i].name, book[i].phone);
	}
}

int main(void)
{
	list_contacts();
	add_contact();
	add_contact();
	list_contacts();

	return 0;
}
```

输入示例（两行，每行一个词）：

```text
xiaoming
13800000001
xiaohong
13800000002
```

<!-- output: 10.5.1 -->
```text
通讯录是空的
请输入姓名：请输入电话：已添加：xiaoming（13800000001）
请输入姓名：请输入电话：已添加：xiaohong（13800000002）
共 2 人：
1. xiaoming 13800000001
2. xiaohong 13800000002
```

输出里两个提示连在一起（`请输入姓名：请输入电话：`）是正常现象：提示先打印、程序在等输入，敲完回车它们才在输出流里"排到一起"（5.5.7 讲过）。

这个里程碑里的检查一个都不能省：

| 检查 | 防什么 |
|------|--------|
| `count >= MAX_CONTACTS` 先拒绝 | 数组越界写（3.1.4） |
| `scanf("%29s", ...)` 限宽 | 缓冲区溢出（3.3.4），`%19s` 同理 |
| `scanf(...)` 返回值 `!= 1` | 输入流关闭或无内容时程序乱跑（1.5.7） |

`add_contact` 返回 `int` 报告成败——函数用返回值说话，调用者决定怎么处理，这是第 5 章定的纪律。

## 10.5.4 里程碑三：存盘与读盘

数据现在只活在内存里，程序一退出就没了。第 5 章的文件读写让数据"关掉再开还在"。文件格式沿用 5.5 定下的：第一行是人数，之后每行"姓名 电话"：

```text
2
xiaoming 13800000001
xiaohong 13800000002
```

```c
// 10.5.3 contact_milestone3.c
#include <stdio.h>
#include <string.h>

#define MAX_CONTACTS 5
#define NAME_LEN 30
#define PHONE_LEN 20
#define DATA_FILE "contacts_book.txt"

struct contact {
	char name[NAME_LEN];
	char phone[PHONE_LEN];
};

struct contact book[MAX_CONTACTS];
int count = 0;

int save_contacts(void)
{
	FILE *fp = fopen(DATA_FILE, "w");

	if (fp == NULL) {
		return -1;
	}

	fprintf(fp, "%d\n", count);
	for (int i = 0; i < count; i++) {
		fprintf(fp, "%s %s\n", book[i].name, book[i].phone);
	}

	fclose(fp);
	return 0;
}

int load_contacts(void)
{
	FILE *fp = fopen(DATA_FILE, "r");

	if (fp == NULL) {
		return -1;
	}

	int loaded = 0;
	if (fscanf(fp, "%d", &loaded) != 1 || loaded < 0 ||
	    loaded > MAX_CONTACTS) {
		fclose(fp);
		return -1;
	}

	for (int i = 0; i < loaded; i++) {
		if (fscanf(fp, "%29s %19s",
			   book[i].name, book[i].phone) != 2) {
			fclose(fp);
			return -1;
		}
	}

	fclose(fp);
	count = loaded;
	return 0;
}

void list_contacts(void)
{
	if (count == 0) {
		printf("通讯录是空的\n");
		return;
	}

	printf("共 %d 人：\n", count);
	for (int i = 0; i < count; i++) {
		printf("%d. %s %s\n", i + 1, book[i].name, book[i].phone);
	}
}

int main(void)
{
	book[0] = (struct contact){"xiaoming", "13800000001"};
	book[1] = (struct contact){"xiaohong", "13800000002"};
	count = 2;

	if (save_contacts() != 0) {
		printf("保存失败\n");
		return 1;
	}
	printf("已保存 %d 人到 %s\n", count, DATA_FILE);

	count = 0;
	printf("清空内存，count = %d\n", count);

	if (load_contacts() != 0) {
		printf("加载失败\n");
		return 1;
	}
	printf("已从 %s 读回 %d 人\n", DATA_FILE, count);
	list_contacts();

	return 0;
}
```

<!-- output: 10.5.3 -->
```text
已保存 2 人到 contacts_book.txt
清空内存，count = 0
已从 contacts_book.txt 读回 2 人
共 2 人：
1. xiaoming 13800000001
2. xiaohong 13800000002
```

先存、再故意把 `count` 清零、再读——模拟"程序重启数据还在"。`load_contacts` 的三道防线是 5.5.3 到 5.5.5 的原样复用，一个都不能少：

1. **`fscanf(fp, "%d", &loaded) != 1`**：文件头不是数字（空文件、被改坏）时拒绝。
2. **`loaded < 0 || loaded > MAX_CONTACTS`**：个数超出范围时拒绝。文件说有 999 人、数组只有 100 格，照单全收就是越界写。
3. **`fscanf(fp, "%29s %19s", ...) != 2`**：每条记录限宽读入，读不齐两条就拒绝，残缺的记录不带进内存。

5.5.6 验证过这些防线真的在干活——被篡改成 `999` 的文件头会被当场拦下。毕业项目把同一套防线原样搬进来，因为文件的来源和那时一样不可全信。

## 10.5.5 里程碑四：函数指针数组分发菜单

四个功能就位，现在搭菜单。用 10.3.4 的函数指针数组做分发，不用一长串 `switch`：

```c
// 10.5.4 contact_menu_demo.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_CONTACTS 3
#define NAME_LEN 30
#define PHONE_LEN 20

struct contact {
	char name[NAME_LEN];
	char phone[PHONE_LEN];
};

struct contact book[MAX_CONTACTS];
int count = 0;

void list_contacts(void)
{
	if (count == 0) {
		printf("通讯录是空的\n");
		return;
	}

	printf("共 %d 人：\n", count);
	for (int i = 0; i < count; i++) {
		printf("%d. %s %s\n", i + 1, book[i].name, book[i].phone);
	}
}

void menu_add(void)
{
	if (count >= MAX_CONTACTS) {
		printf("通讯录已满\n");
		return;
	}

	printf("请输入姓名：");
	if (scanf("%29s", book[count].name) != 1) {
		printf("姓名读取失败\n");
		return;
	}

	printf("请输入电话：");
	if (scanf("%19s", book[count].phone) != 1) {
		printf("电话读取失败\n");
		return;
	}

	printf("已添加：%s（%s）\n", book[count].name, book[count].phone);
	count++;
}

void menu_list(void)
{
	list_contacts();
}

int main(void)
{
	void (*actions[])(void) = {menu_add, menu_list};

	printf("====== 我的通讯录 ======\n");
	printf("1. 添加联系人\n");
	printf("2. 显示全部\n");
	printf("0. 退出\n");

	int choice = 1;
	if (choice >= 1 && choice <= 2) {
		actions[choice - 1]();
	}

	choice = 2;
	if (choice >= 1 && choice <= 2) {
		actions[choice - 1]();
	}

	return 0;
}
```

输入示例：

```text
xiaoming
13800000001
```

<!-- output: 10.5.4 -->
```text
====== 我的通讯录 ======
1. 添加联系人
2. 显示全部
0. 退出
请输入姓名：请输入电话：已添加：xiaoming（13800000001）
共 1 人：
1. xiaoming 13800000001
```

菜单函数统一形状 `void xxx(void)`，数组里按下标排队，`choice - 1` 一句分发。2.2 留过话："菜单循环用 `do-while` 正好"；10.3 演示了函数指针数组；这里两者合体——骨架（打印菜单、读选项、分发）和规则（每个选项干什么）分开了，加功能就是加一个函数加一个数组项。

## 10.5.6 完整版：毕业项目

四个里程碑拼起来，加上删除功能和"退出自动保存"，就是完整的毕业项目。特别说明三个设计决定，它们都是对旧版教程缺陷的修复：

1. **`scanf("%d", &choice)` 检查返回值，失败就清空这一行重来**（`read_choice`）。旧版不检查：输入流被关闭（比如按 Ctrl+D）时 `scanf` 永远失败，`choice` 保持旧值，菜单陷入死循环。新版的处理方式和 1.5.7 一致——返回值不是 1 就把残留输入读光，返回 -1 让菜单提示"没有这个选项"。
2. **退出时自动存盘**。旧版退出时提示"数据已保存在内存中，记得先选 5 存盘哦"——把存盘的责任推给用户，忘了选 5，这次添加的数据就全丢了。新版在用户选 0 时自动调用 `save_contacts`，要么真的保存并如实报告，要么报告保存失败。**程序说什么，就得做到什么。**
3. **读盘校验文件头**。旧版 `fscanf(fp, "%d", &count)` 不看返回值、不查范围，文件被改坏会直接越界写。新版是 10.5.4 的三道防线。

```c
// 10.5.5 contact_full.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_CONTACTS 100
#define NAME_LEN 30
#define PHONE_LEN 20
#define DATA_FILE "contacts_book.txt"

struct contact {
	char name[NAME_LEN];
	char phone[PHONE_LEN];
};

struct contact book[MAX_CONTACTS];
int count = 0;

/* ---------- 基础功能 ---------- */

void list_contacts(void)
{
	if (count == 0) {
		printf("通讯录是空的\n");
		return;
	}

	printf("共 %d 人：\n", count);
	for (int i = 0; i < count; i++) {
		printf("%d. %s %s\n", i + 1, book[i].name, book[i].phone);
	}
}

void menu_add(void)
{
	if (count >= MAX_CONTACTS) {
		printf("通讯录已满，不能再添加\n");
		return;
	}

	printf("请输入姓名：");
	if (scanf("%29s", book[count].name) != 1) {
		printf("姓名读取失败\n");
		return;
	}

	printf("请输入电话：");
	if (scanf("%19s", book[count].phone) != 1) {
		printf("电话读取失败\n");
		return;
	}

	printf("已添加：%s（%s）\n", book[count].name, book[count].phone);
	count++;
}

void menu_search(void)
{
	char key[NAME_LEN];

	printf("要找谁：");
	if (scanf("%29s", key) != 1) {
		printf("读取失败\n");
		return;
	}

	for (int i = 0; i < count; i++) {
		if (strcmp(book[i].name, key) == 0) {
			printf("找到：%s（%s）\n",
			       book[i].name, book[i].phone);
			return;
		}
	}

	printf("没有找到 %s\n", key);
}

void menu_delete(void)
{
	char key[NAME_LEN];

	printf("要删谁：");
	if (scanf("%29s", key) != 1) {
		printf("读取失败\n");
		return;
	}

	for (int i = 0; i < count; i++) {
		if (strcmp(book[i].name, key) == 0) {
			for (int j = i; j < count - 1; j++) {
				book[j] = book[j + 1];
			}
			count--;
			printf("已删除 %s\n", key);
			return;
		}
	}

	printf("没有找到 %s\n", key);
}

/* ---------- qsort 比较器（10.4） ---------- */

int compare_by_name(const void *a, const void *b)
{
	const struct contact *ca = (const struct contact *)a;
	const struct contact *cb = (const struct contact *)b;

	return strcmp(ca->name, cb->name);
}

void menu_sort(void)
{
	qsort(book, (size_t)count, sizeof(struct contact), compare_by_name);
	printf("已按姓名排序\n");
	list_contacts();
}

/* ---------- 存盘与读盘（第 5 章） ---------- */

int save_contacts(void)
{
	FILE *fp = fopen(DATA_FILE, "w");

	if (fp == NULL) {
		return -1;
	}

	fprintf(fp, "%d\n", count);
	for (int i = 0; i < count; i++) {
		fprintf(fp, "%s %s\n", book[i].name, book[i].phone);
	}

	fclose(fp);
	return 0;
}

int load_contacts(void)
{
	FILE *fp = fopen(DATA_FILE, "r");

	if (fp == NULL) {
		return -1;
	}

	int loaded = 0;
	if (fscanf(fp, "%d", &loaded) != 1 || loaded < 0 ||
	    loaded > MAX_CONTACTS) {
		fclose(fp);
		return -1;
	}

	for (int i = 0; i < loaded; i++) {
		if (fscanf(fp, "%29s %19s",
			   book[i].name, book[i].phone) != 2) {
			fclose(fp);
			return -1;
		}
	}

	fclose(fp);
	count = loaded;
	return 0;
}

void menu_save(void)
{
	if (save_contacts() != 0) {
		printf("保存失败，请检查磁盘和权限\n");
		return;
	}
	printf("已保存 %d 人到 %s\n", count, DATA_FILE);
}

void menu_load(void)
{
	if (load_contacts() != 0) {
		printf("加载失败（文件不存在或内容不合法）\n");
		return;
	}
	printf("已从 %s 读入 %d 人\n", DATA_FILE, count);
	list_contacts();
}

/* ---------- 菜单主循环 ---------- */

void show_menu(void)
{
	printf("\n====== 我的通讯录 ======\n");
	printf("1. 添加联系人\n");
	printf("2. 显示全部\n");
	printf("3. 按姓名查找\n");
	printf("4. 按姓名排序\n");
	printf("5. 删除联系人\n");
	printf("6. 保存到文件\n");
	printf("7. 从文件读取\n");
	printf("0. 退出\n");
}

int read_choice(void)
{
	int choice = -1;

	if (scanf("%d", &choice) != 1) {
		int ch;
		while ((ch = getchar()) != '\n' && ch != EOF) {
		}
		return -1;
	}
```

## 10.5.8 常见错误

| 错误写法 | 问题 | 正确写法 |
|----------|------|----------|
| `scanf("%s", name)` 读姓名 | 不限宽，超长输入溢出 | `scanf("%29s", name)` 或 `fgets` |
| `scanf("%d", &choice)` 不查返回值 | 输入流关闭时死循环 | 检查返回值，失败清行重试 |
| 读盘不校验文件头的个数 | 被改坏的文件引发越界写 | 返回值 + 范围两道检查 |
| 退出时只提示"记得存盘" | 提示和实际行为不一致，用户忘存就丢数据 | 退出前自动 `save_contacts` 并如实报告 |
| `actions[choice]()` 不查范围 | 选项越界读函数指针数组 | 先验证 `0 <= choice <= 7` 再分发 |
| 删除后忘了 `count--` | 显示出现重复项，存盘写出垃圾数据 | 删除即 `count--`，两者绑定 |

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NAME_LEN 30
#define PHONE_LEN 20

struct contact {
	char name[NAME_LEN];
	char phone[PHONE_LEN];
};

int compare_by_name(const void *a, const void *b)
{
	const struct contact *ca = (const struct contact *)a;
	const struct contact *cb = (const struct contact *)b;

	return strcmp(ca->name, cb->name);
}

int main(void)
{
	struct contact book[3];

	for (int i = 0; i < 3; i++) {
		printf("请输入第 %d 位联系人的姓名：", i + 1);
		if (scanf("%29s", book[i].name) != 1) {
			printf("读取失败\n");
			return 1;
		}
		printf("请输入电话：");
		if (scanf("%19s", book[i].phone) != 1) {
			printf("读取失败\n");
			return 1;
		}
	}

	qsort(book, 3, sizeof(struct contact), compare_by_name);

	for (int i = 0; i < 3; i++) {
		printf("%s %s\n", book[i].name, book[i].phone);
	}

	FILE *fp = fopen("contacts.txt", "w");
	if (fp == NULL) {
		printf("保存失败\n");
		return 1;
	}
	fprintf(fp, "%d\n", 3);
	for (int i = 0; i < 3; i++) {
		fprintf(fp, "%s %s\n", book[i].name, book[i].phone);
	}
	fclose(fp);
	printf("已保存 3 人\n");

	return 0;
}
```
