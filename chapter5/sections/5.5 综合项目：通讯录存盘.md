# 5.5 综合项目：通讯录存盘

学完打开、读写、定位，这一节把它们组装成一个完整的小项目：**通讯录**——数据放在结构体数组里，存进文本文件，程序重启还能读回来。这个项目同时把本章所有"防错"习惯用上：检查返回值、限制读入宽度、校验文件内容。

## 5.5.1 项目设计

数据格式先定下来。用文本格式存，人可以直接打开看，也天然可移植（5.4.3 讲过二进制文件的移植问题）。文件第一行写联系人的**个数**，后面每行一个联系人：姓名、电话，用空格隔开。

```text
2
xiaoming 13800000001
xiaohong 13800000002
```

内存和文件各管一半：内存里的通讯录像你脑子里的名单，随手翻、随手改；文件像抄在笔记本上的正式名单，关机也不丢。程序要做的事就是在"脑子"和"笔记本"之间来回抄：

| 角色 | 用什么 |
|------|--------|
| 内存里的通讯录 | `struct contact` 数组 + 个数 `count` |
| 硬盘上的通讯录 | `contacts.txt`，第一行是个数 |
| 存盘函数 `save_contacts` | 把数组写进文件 |
| 读盘函数 `load_contacts` | 把文件读回数组 |

`main` 只负责组织：准备数据、存、读、展示。读写细节全在两个函数里，这就是第 2 章讲的分工。

## 5.5.2 保存与加载的完整实现

```c
// 5.5.1 contact_save_load.c
#include <stdio.h>

#define MAX_CONTACTS 100
#define DATA_FILE "contacts.txt"

struct contact {
	char name[30];
	char phone[20];
};

int save_contacts(const struct contact list[], int count)
{
	FILE *fp = fopen(DATA_FILE, "w");
	if (fp == NULL) {
		return -1;
	}

	fprintf(fp, "%d\n", count);
	for (int i = 0; i < count; i++) {
		fprintf(fp, "%s %s\n", list[i].name, list[i].phone);
	}

	fclose(fp);
	return 0;
}

int load_contacts(struct contact list[], int max)
{
	FILE *fp = fopen(DATA_FILE, "r");
	if (fp == NULL) {
		return -1;
	}

	int count = 0;
	if (fscanf(fp, "%d", &count) != 1 || count < 0 || count > max) {
		fclose(fp);
		return -1;
	}

	for (int i = 0; i < count; i++) {
		if (fscanf(fp, "%29s %19s", list[i].name, list[i].phone) != 2) {
			fclose(fp);
			return -1;
		}
	}

	fclose(fp);
	return count;
}

int main(void)
{
	struct contact book[MAX_CONTACTS] = {
		{"xiaoming", "13800000001"},
		{"xiaohong", "13800000002"},
		{"xiaogang", "13800000003"},
	};
	int count = 3;

	if (save_contacts(book, count) != 0) {
		printf("保存失败\n");
		return 1;
	}
	printf("已保存 %d 个联系人\n", count);

	struct contact loaded[MAX_CONTACTS];
	int loaded_count = load_contacts(loaded, MAX_CONTACTS);
	if (loaded_count < 0) {
		printf("加载失败\n");
		return 1;
	}

	printf("从文件读回 %d 个联系人：\n", loaded_count);
	for (int i = 0; i < loaded_count; i++) {
		printf("%s：%s\n", loaded[i].name, loaded[i].phone);
	}

	return 0;
}
```

```text
已保存 3 个联系人
从文件读回 3 个联系人：
xiaoming：13800000001
xiaohong：13800000002
xiaogang：13800000003
```

先记住整体骨架，再看三个安全细节，它们每一个都在防一类真实事故。

## 5.5.3 安全细节一：文件头的 count 必须校验

`load_contacts` 里最关键的一行是：

```c
if (fscanf(fp, "%d", &count) != 1 || count < 0 || count > max) {
	fclose(fp);
	return -1;
}
```

一个条件、三层防线的组合：

1. **`!= 1`**：`fscanf` 必须成功读到 1 个整数。文件是空的、开头是一堆字母，都会在这里被拦下。
2. **`count < 0`**：个数为负没有意义。带着负数继续，后面的 `for (int i = 0; i < count; i++)` 不执行还是小事，某些计算里负数会引发更怪的行为。
3. **`count > max`**：文件声称有 999 个联系人，但数组只有 `MAX_CONTACTS` 个位置。没有这道检查，后面的循环会写越界——和 3.1.4 的数组越界是同一类未定义行为，只是这次由文件内容触发。这就像快递单上写着"一共 999 件"，可你家门口的柜子只有 100 格：不核实就照单收货，多出来的件只能砸到门外去。**攻击者篡改的第一行往往就是这种"个数"字段。**

三个条件都不成立，才把 `count` 当真。

## 5.5.4 安全细节二：读字符串必须限宽

```c
fscanf(fp, "%29s %19s", list[i].name, list[i].phone)
```

`name` 是 30 字节，所以写 `%29s`——最多读 29 个字符，给结尾的 `'\0'` 留 1 个字节；`phone` 是 20 字节，写 `%19s`。规则就是：**宽度 = 缓冲区大小 − 1**。这就像座位号：教室 30 个座位，报到第 29 号就得停，最后一个位置是留给"名字到此为止"这个记号的。

不限宽的 `fscanf(fp, "%s %s", ...)` 和 `scanf("%s")` 是同一种错误：文件里出现一段超长的"姓名"，就会冲出数组边界，把相邻内存写坏。第 3 章读键盘时养成的习惯，读文件一个都不能少。

## 5.5.5 安全细节三：加载失败不硬撑

`load_contacts` 在文件打不开、格式不对、条目残缺时统一返回 `-1`，`main` 收到后报告"加载失败"并退出。这个程序的文件是它自己写的，文件坏了说明出了大问题（被人改了、磁盘出错了），硬着头皮用残缺数据比直接失败更危险。

另外注意 `load_contacts` 收的是 `max` 参数而不是写死 100——数组多大多大，函数就该防多大。

## 5.5.6 眼见为实：被篡改的文件头

上面几条防线看着像"理论上"，做一个实验验证它们真的在干活。手工把文件头改成 `999`（远超上限），再看加载函数的反应：

```c
// 5.5.2 bad_count.c
#include <stdio.h>

#define MAX_CONTACTS 100

int main(void)
{
	FILE *fp = fopen("broken.txt", "w");
	if (fp == NULL) {
		return 1;
	}
	fprintf(fp, "999\n");
	fclose(fp);

	fp = fopen("broken.txt", "r");
	if (fp == NULL) {
		return 1;
	}

	int count = 0;
	if (fscanf(fp, "%d", &count) != 1 || count < 0 || count > MAX_CONTACTS) {
		printf("文件头不合法：count 是 %d\n", count);
		printf("拒绝读取，程序安全退出\n");
		fclose(fp);
		return 0;
	}

	fclose(fp);
	printf("读完 %d 个联系人\n", count);

	return 0;
}
```

<!-- output: 5.5.2 -->
```text
文件头不合法：count 是 999
拒绝读取，程序安全退出
```

把 5.5.1 里的校验条件删掉试试：程序会试图读 999 条不存在的记录，在越界写入处变成未定义行为。防线平时看不见，出事时就是它顶上。

## 5.5.7 添加联系人：一个能交互的小工具

最后把项目做成能用的样子：启动时读文件（没有就放两个初始联系人），让用户输入一个新联系人，存盘，再展示全部。

```c
// 5.5.3 contact_add.c
#include <stdio.h>
#include <string.h>

#define MAX_CONTACTS 100
#define DATA_FILE "contacts_add.txt"

struct contact {
	char name[30];
	char phone[20];
};

int save_contacts(const struct contact list[], int count)
{
	FILE *fp = fopen(DATA_FILE, "w");
	if (fp == NULL) {
		return -1;
	}

	fprintf(fp, "%d\n", count);
	for (int i = 0; i < count; i++) {
		fprintf(fp, "%s %s\n", list[i].name, list[i].phone);
	}

	fclose(fp);
	return 0;
}

int load_contacts(struct contact list[], int max)
{
	FILE *fp = fopen(DATA_FILE, "r");
	if (fp == NULL) {
		return -1;
	}

	int count = 0;
	if (fscanf(fp, "%d", &count) != 1 || count < 0 || count > max) {
		fclose(fp);
		return -1;
	}

	for (int i = 0; i < count; i++) {
		if (fscanf(fp, "%29s %19s", list[i].name, list[i].phone) != 2) {
			fclose(fp);
			return -1;
		}
	}

	fclose(fp);
	return count;
}

int main(void)
{
	struct contact book[MAX_CONTACTS];
	int count = load_contacts(book, MAX_CONTACTS);

	if (count < 0) {
		printf("没有数据文件，先放入 2 个初始联系人\n");
		strcpy(book[0].name, "xiaoming");
		strcpy(book[0].phone, "13800000001");
		strcpy(book[1].name, "xiaohong");
		strcpy(book[1].phone, "13800000002");
		count = 2;
	} else {
		printf("已读入 %d 个联系人\n", count);
	}

	if (count >= MAX_CONTACTS) {
		printf("通讯录已满，不能添加\n");
		return 1;
	}

	char name[30];
	char phone[20];

	printf("请输入新联系人的姓名：");
	if (scanf("%29s", name) != 1) {
		printf("姓名读取失败\n");
		return 0;
	}

	printf("请输入新联系人的电话：");
	if (scanf("%19s", phone) != 1) {
		printf("电话读取失败\n");
		return 0;
	}

	strcpy(book[count].name, name);
	strcpy(book[count].phone, phone);
	count++;

	if (save_contacts(book, count) != 0) {
		printf("保存失败\n");
		return 1;
	}

	printf("已保存，通讯录现在有 %d 个联系人：\n", count);
	for (int i = 0; i < count; i++) {
		printf("%s：%s\n", book[i].name, book[i].phone);
	}

	return 0;
}
```

输入示例（姓名一行、电话一行）：

```text
laowang
13900000009
```

<!-- output: 5.5.3 -->
```text
没有数据文件，先放入 2 个初始联系人
请输入新联系人的姓名：请输入新联系人的电话：已保存，通讯录现在有 3 个联系人：
xiaoming：13800000001
xiaohong：13800000002
laowang：13900000009
```

几个值得注意的地方：

1. **行为和提示自洽**。程序说"已保存"，就真的调用了 `save_contacts` 把全部联系人（含新加的）重写进文件。下次启动，`load_contacts` 会读出这 3 个人——数据真的留下了，而不是嘴上说说。
2. **提示和输入的顺序**。输出里两个提示连在一起，是因为提示先打印、程序在等输入，你敲完回车它们才"流"到一起，属正常现象。
3. 交互读入也限宽：`scanf("%29s", name)`、`scanf("%19s", phone)`，和读文件同一套纪律。
4. 添加前检查 `count >= MAX_CONTACTS`，通讯录满了就拒绝，不做越界写入。

## 5.5.8 常见错误

| 错误写法 | 问题 | 正确写法 |
|----------|------|----------|
| `fscanf(fp, "%d", &count);` 不检查就当个数用 | 文件坏了或 count 超大时越界 | `if (fscanf(...) != 1 \|\| count < 0 \|\| count > max) { ... }` |
| `fscanf(fp, "%s %s", a, b);` 不限宽 | 超长内容冲垮数组 | `fscanf(fp, "%29s %19s", a, b)` |
| 读完一条 `scanf("%29s")` 就继续下一条 | 返回值没看，残缺行悄悄吞掉 | 每次都检查 `!= 1` 或 `!= 2` |
| 添加后只改内存、不存盘 | 提示"已保存"但数据没落盘 | 改完立即调用保存函数 |
| `count` 声称的上限和数组实际大小不一致 | 两处写死两个数字，改一个漏一个 | 同一个 `MAX_CONTACTS` 宏到处用 |

## 5.5.9 小结

- 持久化的固定套路：启动时加载文件到数组，修改数组，退出前（或每次修改后）存盘。
- 文本格式 + `count` 头是最简单可靠的自定义存档格式：人能看、可移植、好排查。
- 读文件头必须三连检查：`fscanf` 返回值、下限、上限。
- 读字符串永远限宽，宽度是缓冲区大小减 1。
- 保存失败、加载失败都要有明确的应对，不能带病运行。
- 一个"管理系统"= 内存里的数组 + 硬盘上的文件 + 在两者之间搬运数据的函数。

5. 编程题：给 5.5.1 的通讯录补一个 `find_contact` 函数，按姓名查找，返回下标，找不到返回 -1。用 `strcmp` 比较，输出格式如下（查 `xiaohong` 和 `nobody` 各一次）：

```text
找到：xiaohong 13800000002
没找到：nobody
```

<details>
<summary>参考答案</summary>

1. 一眼就知道文件里有多少条记录，读到几个就停，也不至于把文件尾的垃圾当数据。不存的话得读到文件末尾为止（`fgets`/`fscanf` 返回失败为止），逻辑也能写，但"先给个数"让校验更容易：个数不合理直接拒绝整个文件。
2. `%39s`。宽度是缓冲区大小减 1，40 − 1 = 39。
3. 数组是在 `main` 里定义的，大小由调用处决定。写死 `MAX_CONTACTS` 的话，一旦别处用不同大小定义数组（比如只开 10 个位置），校验就防不住了。跟着传入的 `max` 走，函数对任何大小的数组都安全。
4. 三处问题。第一，`fscanf` 的返回值没检查，读不到个数时 `count` 是未初始化的垃圾值。第二，没校验 `count` 的范围，负数没意义、超过数组大小会越界写入。第三，`"%s %s"` 没有限宽，超长的姓名和电话会冲出数组。正确写法见 5.5.1 的 `load_contacts`。
5. 参考代码：

```c
#include <stdio.h>
#include <string.h>

#define MAX_CONTACTS 100

struct contact {
	char name[30];
	char phone[20];
};

int find_contact(const struct contact list[], int count, const char *name)
{
	for (int i = 0; i < count; i++) {
		if (strcmp(list[i].name, name) == 0) {
			return i;
		}
	}
	return -1;
}

int main(void)
{
	struct contact book[MAX_CONTACTS] = {
		{"xiaoming", "13800000001"},
		{"xiaohong", "13800000002"},
	};
	int count = 2;

	int index = find_contact(book, count, "xiaohong");
	if (index >= 0) {
		printf("找到：%s %s\n", book[index].name, book[index].phone);
	} else {
		printf("没找到：xiaohong\n");
	}

	index = find_contact(book, count, "nobody");
	if (index >= 0) {
		printf("找到：%s %s\n", book[index].name, book[index].phone);
	} else {
		printf("没找到：nobody\n");
	}

	return 0;
}
```
