## 5.5.1 新技能：命令行参数 argc / argv

### 🤔 你有没有想过：main 也可以有参数？

之前的 `main` 都是空括号。其实标准的 `main` 可以接收**用户运行程序时传进来的信息**：

```c
int main(int argc, char *argv[]) {
    // ...
}
```

| 参数 | 含义 |
|------|------|
| `argc` | argument count，参数**个数** |
| `argv` | argument vector，参数**内容**（字符串数组） |

### 📝 举个例子

```c
#include <stdio.h>

int main(int argc, char *argv[]) {
    printf("参数个数：%d\n", argc);
    
    for (int i = 0; i < argc; i++) {
        printf("第 %d 个参数：%s\n", i, argv[i]);
    }
    
    return 0;
}
```

**运行：**

```bash
gcc args.c -o args
./args hello world 123
```

**输出：**
```
参数个数：4
第 0 个参数：./args
第 1 个参数：hello
第 2 个参数：world
第 3 个参数：123
```

> 📌 **规律：**
> - `argv[0]` 永远是**程序自己**的名字（`./args`）
> - 从 `argv[1]` 开始才是你传的参数
> - `argc` 就是参数的个数（含程序名）
> 
> 以后我们就能像 `./program data.txt` 这样，把文件名告诉程序啦！

## 5.5.2 项目目标：学生成绩管理系统（文件版）

我们来做这样一个程序，功能如下：

```
1. 程序启动时，从文件 scores.txt 读取学生成绩
2. 打印所有学生
3. 计算并打印平均分、最高分
4. 添加一个学生（写入内存 + 同步保存到文件）
```

**关键卖点：数据存在文件里，程序关掉再开，数据还在！**

---

## 5.5.3 第一步：定义数据和文件函数

```c
#include <stdio.h>
#include <string.h>

#define MAX_STUDENTS 100    // 最多100个学生
#define DATA_FILE "scores.txt"

typedef struct {
    char name[20];
    double score;
} Student;

// 函数声明
int loadFromFile(Student arr[], int maxCount);        // 读文件 → 返回人数
int saveToFile(Student arr[], int count);             // 存文件 → 成功0/失败-1
void printAll(Student arr[], int count);
double averageScore(Student arr[], int count);
int topIndex(Student arr[], int count);
```

> 📌 **`#define`** 是第1章提过的"宏定义"，`#define MAX_STUDENTS 100` 表示：代码里的 `MAX_STUDENTS` 都会替换成 100。用名字比到处写死数字好——想改时只改一行！

---

## 5.5.4 第二步：实现文件读写函数

### 📝 保存：把整个数组写进文件

每行存一个学生：`名字 分数`。用文本方式，方便人看：

```c
int saveToFile(Student arr[], int count) {
    FILE *fp = fopen(DATA_FILE, "w");    // "w" 清空重写整个文件
    if (fp == NULL) {
        perror("保存失败");
        return -1;
    }
    
    for (int i = 0; i < count; i++) {
        fprintf(fp, "%s %.1f\n", arr[i].name, arr[i].score);
    }
    
    fclose(fp);
    return 0;
}
```

### 📝 加载：从文件读回数组

```c
int loadFromFile(Student arr[], int maxCount) {
    FILE *fp = fopen(DATA_FILE, "r");
    
    // 文件不存在不算"致命错误"——就当是空系统，从0开始
    if (fp == NULL) {
        printf("没有找到数据文件，将从头开始。\n");
        return 0;
    }
    
    int count = 0;
    // 逐行读：%s 读名字，%lf 读分数；成功读到2个才继续
    while (count < maxCount &&
           fscanf(fp, "%s %lf", arr[count].name, &arr[count].score) == 2) {
        count++;
    }
    
    fclose(fp);
    printf("成功从文件载入 %d 个学生。\n", count);
    return count;
}
```

> 💡 **注意 `loadFromFile` 的容错设计：**
> - 文件不存在 → 返回 0（当作"还没有学生"），**不是错误**
> - 用 `count < maxCount` 防止把数组撑爆（防御式编程！）

## 5.5.5 第三步：统计与显示函数

```c
void printAll(Student arr[], int count) {
    if (count == 0) {
        printf("（暂无学生）\n");
        return;
    }
    for (int i = 0; i < count; i++) {
        printf("%s：%.1f分\n", arr[i].name, arr[i].score);
    }
}

double averageScore(Student arr[], int count) {
    if (count == 0) {
        return 0;
    }
    double sum = 0;
    for (int i = 0; i < count; i++) {
        sum += arr[i].score;
    }
    return sum / count;
}

int topIndex(Student arr[], int count) {
    int top = 0;
    for (int i = 1; i < count; i++) {
        if (arr[i].score > arr[top].score) {
            top = i;
        }
    }
    return top;
}
```

## 5.5.6 第四步：组装 main（完整代码）

把上面的函数和 main 拼在一起，就是完整程序：

```c
#include <stdio.h>
#include <string.h>

#define MAX_STUDENTS 100
#define DATA_FILE "scores.txt"

typedef struct {
    char name[20];
    double score;
} Student;

// 函数声明
int loadFromFile(Student arr[], int maxCount);
int saveToFile(Student arr[], int count);
void printAll(Student arr[], int count);
double averageScore(Student arr[], int count);
int topIndex(Student arr[], int count);

int main(void) {
    Student class[MAX_STUDENTS];
    
    // 1. 启动时载入文件里保存的数据
    int count = loadFromFile(class, MAX_STUDENTS);
    
    // 2. 显示当前所有学生
    printf("\n=== 当前学生名单 ===\n");
    printAll(class, count);
    
    // 3. 统计
    if (count > 0) {
        int top = topIndex(class, count);
        printf("\n平均分：%.1f\n", averageScore(class, count));
        printf("最高分：%s（%.1f分）\n", class[top].name, class[top].score);
    }
    
    // 4. 演示"添加学生并保存"
    // （真实程序这里会用 scanf 让用户输入，这里用固定数据演示）
    strcpy(class[count].name, "新同学");
    class[count].score = 88.5;
    count++;
    
    if (saveToFile(class, count) == 0) {
        printf("\n已保存 %d 个学生到 %s\n", count, DATA_FILE);
    }
    
    // 5. 再显示一遍，证明新学生已加入
    printf("\n=== 保存后的名单 ===\n");
    printAll(class, count);
    
    return 0;
}

// ---- 以下是函数实现 ----

int saveToFile(Student arr[], int count) {
    FILE *fp = fopen(DATA_FILE, "w");
    if (fp == NULL) {
        perror("保存失败");
        return -1;
    }
    for (int i = 0; i < count; i++) {
        fprintf(fp, "%s %.1f\n", arr[i].name, arr[i].score);
    }
    fclose(fp);
    return 0;
}

int loadFromFile(Student arr[], int maxCount) {
    FILE *fp = fopen(DATA_FILE, "r");
    if (fp == NULL) {
        printf("没有找到数据文件，将从头开始。\n");
        return 0;
    }
    int count = 0;
    while (count < maxCount &&
           fscanf(fp, "%s %lf", arr[count].name, &arr[count].score) == 2) {
        count++;
    }
    fclose(fp);
    printf("成功从文件载入 %d 个学生。\n", count);
    return count;
}

void printAll(Student arr[], int count) {
    if (count == 0) {
        printf("（暂无学生）\n");
        return;
    }
    for (int i = 0; i < count; i++) {
        printf("%s：%.1f分\n", arr[i].name, arr[i].score);
    }
}

double averageScore(Student arr[], int count) {
    if (count == 0) {
        return 0;
    }
    double sum = 0;
    for (int i = 0; i < count; i++) {
        sum += arr[i].score;
    }
    return sum / count;
}

int topIndex(Student arr[], int count) {
    int top = 0;
    for (int i = 1; i < count; i++) {
        if (arr[i].score > arr[top].score) {
            top = i;
        }
    }
    return top;
}
```

## 5.5.8 本节小结

这一节我们学了：

✅ **命令行参数：** `int main(int argc, char *argv[])`，`argv[0]` 是程序名

✅ **持久化思路：** 启动时"载入"文件 → 运行时修改内存 → 退出前"保存"文件

✅ **综合运用：** 结构体 + 数组 + 循环 + 函数 + 错误处理 + 文件读写

> 💡 **记住：一个"管理系统" = 内存里的数组 + 硬盘上的文件 + 在两者之间搬运数据的函数！**

### 三、改错题

**6. 以下程序想把一个学生数组存盘并读回，有4个错误，请找出并改正：**

```c
#include <stdio.h>

typedef struct { char name[20]; int score; } Student;

int main() {
    Student list[2] = {{"小明", 90}, {"小红", 85}};
    
    FILE *fp = fopen("stu.bin", "w");    // 错误区①：存结构体应该用二进制模式
    
    fwrite(list, sizeof(Student), 2, fp);   // 错误区②：没检查 fp 和返回值
    
    fclose(fp);
    
    fp = fopen("stu.bin", "r");            // 错误区③：读取模式也应二进制
    
    Student loaded[2];
    fread(loaded, sizeof(Student), 2, fp);  // 错误区④：?（这一行本身OK，请检查文件路径）
    
    fclose(fp);
    
    printf("%s %d\n", loaded[0].name, loaded[0].score);
    return 0;
}
```

> 提示：核心是"存结构体要用二进制模式 `wb`/`rb`"，以及 fopen 之后要检查是否 NULL。
